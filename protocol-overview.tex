\def\dsfHs{\dsfi1 Measure $S^{ab}(\bq,\omega)$ at several high $T$.
  \dsfi2 Integrate data to moments $m^{ab}_{n,\bd}$.
  \dsfi3 Extract by finite differences in $\beta$:
  \dsfe{m^{ab}_{n,\bd}(\beta)=\sum_{i\geq0}\beta^i\,\sm^{ab}_{n,\bd,i}.}
  \dsfi4 Use operator algebra to derive $\sm^{ab}_{n,\bd,i}(\bl)$ from proposed Hamiltonian.
  \dsfi5 Equate and solve for $\bl$.}
\def\dsfSs{\dsfi1 Measure $S^{ab}(\bq,\omega)$ in strong fields $B\hat n$.
  \dsfi2 Integrate data to moments $m^{ab}_{n,\bd}$.
  \dsfi3 Extract by finite differences in $1/B$:
  \dsfe{B^{-n}m^{ab}_{n,\bd}(B)=\sum_{i\geq0}B^{-i}\,\sm^{ab}_{n,\bd,i}.}
  \dsfi4 Use perturbation theory to derive $\sm^{ab}_{n,\bd,i}(\bl)$ from proposed Hamiltonian.
  \dsfi5 Equate and solve for $\bl$.}
\def\dsfFs{\dsfi1 Measure $S^{ab}(\bq,\omega)$ at a single $T$.
  \dsfi2 Integrate data to $m^{ab}_{0,\bd}$ and $m^{ab}_{1,\bd}$.
  \dsfi3 Use measured data to form $\mathsf A$ and $\mathsf b$:
  \dsfe{\jot.6pt\begin{aligned}\mathsf A^{ab}_{\bd,j}&=\langle[O^b_{\bd},[H^j,O^a_{\boldsymbol0}]]\rangle_\beta = f(m_{0,\vec d}^{ab}),\\
    \mathsf b^{ab}_{\bd}&=(m^{ab}_{1,\bd}+m^{ba}_{1,-\bd})/2\pi.\end{aligned}}
  \dsfi4 Solve $\sum_j\mathsf A^{ab}_{\bd,j}\,\lambda^j=\mathsf b^{ab}_{\bd}$.}
\def\dsfHg{\dsfl{${\leq}2$-body guarantee}full $S^{ab}$ gives all couplings linearly:
  $\lambda^{ab}_{\bd}=-\sm^{ab}_{0,\bd,1}/2\pi$, $b^a=\mathrm i\varepsilon^{abc}\sm^{bc}_{0,\boldsymbol0,1}/4\pi$.
  \dsfl{Extensions}diagonal $S^{aa}$, higher-body $H$, spatial disorder}
\def\dsfSg{\dsfl{${\leq}2$-body guarantee}full $S^{ab}$ at $\hat n=\hat x,\hat y,\hat z$ gives all
  couplings linearly.
  \dsfl{Extensions}diagonal $S^{aa}$, higher body $H$, angular tomography.}
\def\dsfFg{\dsfl{${\leq}2$-body guarantee}full $S^{ab}$ gives all couplings exactly and linearly if $\det\,m^{ab}_{0,\bd} \neq0$ , $\langle\boldsymbol\sigma\rangle_\beta\neq\boldsymbol0$. \par
\textit{Extensions:} any stationary state}
